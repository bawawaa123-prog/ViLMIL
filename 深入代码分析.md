```python
class PatchesDataset(Dataset):
    def __init__(self, file_path, transform=None):
        # 扫描patch目录中的所有图像文件
        file_names = os.listdir(file_path)
        self.imgs = [os.path.join(file_path, fn) for fn in file_names]
        self.coords = file_names  # 保存原始文件名用于解析坐标
        self.transform = transform

    def __getitem__(self, index):
        # 打开图像、转RGB、应用预处理
        img = Image.open(fn).convert('RGB')
        coord = self.coords[index]
        if self.transform is not None:
            img = self.transform(img)
        return img, coord  # 返回(张量, 文件名字符串)
```

```python
def save_embeddings_biomedclip(model, fname, dataloader, overwrite=False):
    embeddings = []
    coords = []
    
    for batch, coord in dataloader:  # batch: [B, 3, 224, 224]
        with torch.no_grad():
            batch = batch.to(device)
            # BiomedCLIP的图像编码器
            # 输入: [B, 3, 224, 224]

    coords = parse_coords_from_filenames(file_names)  # [N_patches, 2]
```

```python
def get_optim(model, args):
    base_lr = float(getattr(args, 'lr', 1e-4))
    weight_decay = float(getattr(args, 'reg', 0.0))
    
    if args.model_type == 'ViLa_MIL_BiomedCLIP':
        # 参数分组策略
        param_groups = []
        
        # 1️⃣ 提示词学习器 (高学习率)
        if hasattr(model, 'prompt_learner'):
            prompt_lr = getattr(args, 'prompt_lr', min(max(base_lr * 10.0, 1e-4), 1e-3))
            param_groups.append({
                'name': 'prompt_learner',
                'params': model.prompt_learner.parameters(),
                'lr': prompt_lr
            })
        
        # 2️⃣ 文本编码器 (低学习率或冻结)
        if hasattr(model, 'text_encoder'):
            if model.text_encoder.finetune:
                text_lr = getattr(args, 'text_lr', min(base_lr, 1e-5))
                param_groups.append({
                    'name': 'text_encoder',
                    'params': [p for p in model.text_encoder.parameters() if p.requires_grad],
                    'lr': text_lr
                })
        
        # 3️⃣ 其他参数 (基础学习率)
        other_params = [p for p in model.parameters() 
                       if p not in prompt_params and p not in text_params]
        if other_params:
            param_groups.append({
                'name': 'other',
                'params': other_params,
                'lr': base_lr
            })
        
        optimizer = optim.Adam(param_groups, weight_decay=weight_decay)
    
    else:
        optimizer = optim.Adam(model.parameters(), lr=base_lr, weight_decay=weight_decay)
    
    return optimizer
```

**学习率策略**：
- **Prompt Learner**: 10倍基础学习率 (1e-4 → 1e-3)
- **文本编码器**: 1/10基础学习率 (1e-4 → 1e-5)
- **其他模块**: 基础学习率 (1e-4)

```python
def get_split_loader(split_dataset, training=False, testing=False, 
                     weighted=False, mode='clam'):
    
    if mode == 'transformer':
        # Dual-scale模式：同时加载低和高分辨率特征
        collate = collate_tranformer
    
    # Windows兼容性处理
    worker_count = 0 if os.name == 'nt' else 4
    
    if not testing:
        if training:
            if weighted:
                # 平衡类别采样
                weights = make_weights_for_balanced_classes_split(split_dataset)
                loader = DataLoader(
                    split_dataset,
                    batch_size=1,  # ⚠️ MIL固定为1
                    sampler=WeightedRandomSampler(weights, len(weights)),
                    collate_fn=collate,
                    num_workers=worker_count
                )
            else:
                loader = DataLoader(
                    split_dataset,
                    batch_size=1,
                    sampler=RandomSampler(split_dataset),
                    collate_fn=collate,
                    num_workers=worker_count
                )
        else:
            # 验证模式：顺序采样
            loader = DataLoader(
                split_dataset,
                batch_size=1,
                sampler=SequentialSampler(split_dataset),
                collate_fn=collate,
                num_workers=worker_count
            )
    
    return loader

def collate_tranformer(batch):
    """Collate函数：处理batch"""
    img_s = torch.cat([item[0] for item in batch], dim=0)   # 低分辨率特征
    coord_s = torch.cat([item[1] for item in batch], dim=0) # 低分辨率坐标
    img_l = torch.cat([item[2] for item in batch], dim=0)   # 高分辨率特征
    coord_l = torch.cat([item[3] for item in batch], dim=0) # 高分辨率坐标
    label = torch.LongTensor([item[4] for item in batch])   # 标签
    slide_ids = [item[5] for item in batch]                 # 幻灯片ID
    
    return [img_s, coord_s, img_l, coord_l, label, slide_ids]
```

```python
class ViLa_MIL_BiomedCLIP(nn.Module):
    def __init__(self, config, num_classes=2):
        super().__init__()
        
        # ========== 特征维度 ==========

        self.attention_U = nn.Sequential(nn.Linear(512, 192), nn.Sigmoid())
        self.attention_weights = nn.Linear(192, 1)
        
        # 2️⃣ 文本编码器 (BiomedCLIP PubMedBERT)
        self.text_encoder = BiomedCLIPTextEncoder(
            biomedclip_model, n_ctx=16, finetune=False

        # 3️⃣ 可学习提示词
```

### 2. 前向传播流程

#### 文件：`models/model_ViLa_MIL_BiomedCLIP.py:forward()`

```python
def forward(self, x_s, coord_s, x_l, coords_l, label, slide_id=None):
    """
    输入:
        x_s: 低分辨率特征 [1, N_s, 512]
        coord_s: 低分辨率坐标
        x_l: 高分辨率特征 [1, N_l, 512]
        coords_l: 高分辨率坐标
        label: 标签 [1]
        slide_id: 幻灯片ID (用于RAG)
    
    返回:
        Y_prob: 预测概率 [1, num_classes]
        Y_hat: 预测类别
        loss: 交叉熵损失
    """
    
    # ═══════════════════════════════════════════
    # 🔵 阶段1：低分辨率分支处理
    # ═══════════════════════════════════════════
    
    M = x_s.float()  # [1, N_s, 512]
    
    # 1️⃣ 原型指导的patch解码
    # 使用可学习原型作为查询与patch进行交叉注意力
    components, _ = self.cross_attention_1(
        self.learnable_image_center,  # Query: [16, 1, 512]
        M,                             # Key/Value: [1, N_s, 512]
        M
    )  # Output: [16, 1, 512]
    
    # 应用LayerNorm并残差连接
    components = self.norm(components + self.learnable_image_center)
    # Output: [16, 1, 512]
    
    # 2️⃣ MIL注意力聚合
    H = components.squeeze().float()  # [16, 512]
    
    A_V = self.attention_V(H)  # [16, 192]  (Tanh激活)
    A_U = self.attention_U(H)  # [16, 192]  (Sigmoid激活)
    
    A = self.attention_weights(A_V * A_U)  # [16, 1]
    A = torch.transpose(A, 1, 0)            # [1, 16]
    A = F.softmax(A, dim=1)                 # 归一化注意力权重
    
    # 3️⃣ 聚合得到低分辨率图像特征
    image_features_low = torch.mm(A, H)  # [1, 512]
    
    # ═══════════════════════════════════════════
    # 🟢 阶段2：高分辨率分支处理 (完全相同)
    # ═══════════════════════════════════════════
    
    M_high = x_l.float()  # [1, N_l, 512]
    
    components_high, _ = self.cross_attention_1(
        self.learnable_image_center,
        M_high,
        M_high
    )
    components_high = self.norm(components_high + self.learnable_image_center)
    
    H_high = components_high.squeeze().float()  # [16, 512]
    A_V_high = self.attention_V(H_high)
    A_U_high = self.attention_U(H_high)
    A_high = self.attention_weights(A_V_high * A_U_high)
    A_high = torch.transpose(A_high, 1, 0)
    A_high = F.softmax(A_high, dim=1)
    
    image_features_high = torch.mm(A_high, H_high)  # [1, 512]
    
    # ═══════════════════════════════════════════
    # 🟡 阶段3：文本编码 (提示词学习)
    # ═══════════════════════════════════════════
    
    # 生成可学习的提示词嵌入
    prompt_embeddings = self.prompt_learner()  # [num_classes, seq_len, 768]
    tokenized_prompts = self.prompt_learner.tokenized_prompts.to(x_s.device)
    eos_indices = self.prompt_learner.eos_indices.to(x_s.device)
    
    # 使用BiomedCLIP文本编码器 (支持提示词学习)
    # 输出: [2*num_classes, 512]
    #   前num_classes是低分辨率文本特征
    #   后num_classes是高分辨率文本特征
    text_features = self.text_encoder(
        tokenized_prompts,
        prompt_embeddings=prompt_embeddings,
        eos_indices=eos_indices
    )
    
    static_text_low = text_features[:self.num_classes]    # [num_classes, 512]
    static_text_high = text_features[self.num_classes:]   # [num_classes, 512]
    
    # ═══════════════════════════════════════════
    # 🟣 阶段4：动态提示检索 (可选)
    # ═══════════════════════════════════════════
    
    if self.enable_dynamic_prompt and self.dynamic_prompt_runtime_enabled:
        # 从句子池中检索与当前图像最相似的K个提示词
        low_pool_feats, high_pool_feats = self._get_dynamic_pool_features(x_s.device)
        
        text_features_low, text_features_high, retrieval_debug = self.dynamic_retriever(
            low_query=image_features_low,
            high_query=image_features_high,
            low_pool_feats_by_class=low_pool_feats,
            high_pool_feats_by_class=high_pool_feats,
            static_low=static_text_low,
            static_high=static_text_high,
            low_pool_texts_by_class=self._dynamic_low_pool_texts,
            high_pool_texts_by_class=self._dynamic_high_pool_texts
        )
        # 输出: [num_classes, 512] (检索融合后的特征)
    else:
        text_features_low = static_text_low
        text_features_high = static_text_high
        retrieval_debug = None
    
    # ═══════════════════════════════════════════
    # 🔴 阶段5：RAG文本重写 (可选)
    # ═══════════════════════════════════════════
    
    if self.enable_rag_rewrite and self.rag_runtime_enabled and slide_id:
        # 从RAG缓存或在线模型获取重写的提示词
        rag_item = self.rag_rewriter.get_rewritten_prompts(
            str(slide_id),
            retrieval_debug=retrieval_debug
        )
        
        if rag_item:
            low_texts = rag_item.get('low', [])
            high_texts = rag_item.get('high', [])
            
            # 编码RAG生成的文本
            rag_low = self._encode_sentence_list(low_texts).to(x_s.device)
            rag_high = self._encode_sentence_list(high_texts).to(x_s.device)
            
            # 验证形状匹配，否则回退
            if rag_low.shape == text_features_low.shape:
                text_features_low = rag_low
                text_features_high = rag_high
            else:
                # 回退策略：使用动态或静态特征
                text_features_low = text_features_low  # 保持动态/静态
                text_features_high = text_features_high
    
    # ═══════════════════════════════════════════
    # 🟠 阶段6：视觉条件提示(VCP) (可选)
    # ═══════════════════════════════════════════
    
    if self.enable_vcp and self.vcp_runtime_enabled:
        # 基于图像特征动态调整文本特征
        delta_low = self.low_vcp_head(image_features_low)  # [1, 512]
        delta_high = self.high_vcp_head(image_features_high)
        
        # 应用自适应参数化的文本特征修正
        text_features_low = self.vcp_ln_low(
            text_features_low + self.vcp_beta_low * delta_low
        )
        text_features_high = self.vcp_ln_high(
            text_features_high + self.vcp_beta_high * delta_high
        )
    
    # ═══════════════════════════════════════════
    # 🟤 阶段7：文本-图像对齐
    # ═══════════════════════════════════════════
    
    # 构建图像上下文 = 原型 + 原始patch
    image_context = torch.cat((components.squeeze(), M.squeeze(0)), dim=0)
    # [16+N_s, 512]
    
    # 使用文本特征查询图像上下文
    text_context_features, _ = self.cross_attention_2(
        text_features_low.unsqueeze(1),  # Query: [num_classes, 1, 512]
        image_context,                    # Key/Value: [16+N_s, 512]
        image_context
    )  # Output: [num_classes, 1, 512]
    
    # 残差连接
    text_features_low = text_context_features.squeeze() + text_features_low
    
    # 高分辨率同样处理
    image_context_high = torch.cat((components_high.squeeze(), M_high.squeeze(0)), dim=0)
    text_context_features_high, _ = self.cross_attention_2(
        text_features_high.unsqueeze(1),
        image_context_high,
        image_context_high
    )
    text_features_high = text_context_features_high.squeeze() + text_features_high
    
    # ═══════════════════════════════════════════
    # 🔵 阶段8：视觉-文本对齐和分类
    # ═══════════════════════════════════════════
    
    # 计算图像-文本相似度 (双重缩放)
    logits_low = image_features_low @ text_features_low.T    # [1, num_classes]
    logits_high = image_features_high @ text_features_high.T # [1, num_classes]
    
    # 双重缩放融合
    logits = logits_low + logits_high  # [1, num_classes]
    
    # 计算损失
    loss = self.loss_ce(logits, label)
    
    # 获取预测
    Y_prob = F.softmax(logits, dim=1)  # [1, num_classes]
    Y_hat = torch.topk(Y_prob, 1, dim=1)[1]  # [1, 1]
    
    return Y_prob, Y_hat, loss
```

### 3. 关键组件详解

#### A. 提示词学习器 (BiomedCLIPPromptLearner)

```python
class BiomedCLIPPromptLearner(nn.Module):
    def __init__(self, classnames, biomedclip_model, tokenizer, n_ctx=16):
        # classnames: ['adenocarcinoma', 'squamous_cell_carcinoma', ...]
        
        # 获取嵌入维度 (768 for PubMedBERT)
        ctx_dim = 768
        
        # 可学习的上下文向量
        ctx_vectors = torch.empty(n_ctx, ctx_dim)
        nn.init.normal_(ctx_vectors, std=0.02)
        self.ctx = nn.Parameter(ctx_vectors)  # [16, 768]
        
        # 预处理类别名称
        classnames = [name.replace("_", " ") for name in classnames]
        
        # 创建提示词模板
        prompts = [f"a histopathology image of {name}" for name in classnames]
        # 例如: "a histopathology image of adenocarcinoma"
        
        # Tokenize提示词
        self.tokenized_prompts = tokenizer(prompts)  # [num_classes, max_seq_len]
        
        # 提取token嵌入
        with torch.no_grad():
            embedding = text_model.transformer.embeddings.word_embeddings(token_ids)
        
        # 分离前缀([SOS])、后缀(类名+[EOS])
        self.token_prefix = embedding[:, :1, :]              # [num_classes, 1, 768]
        self.token_suffix = embedding[:, 1:1+suffix_keep, :] # [num_classes, suffix_len, 768]
    
    def forward(self):
        # 拼接 [SOS] + learnable_ctx + class_name + [EOS]
        prompts = torch.cat([
            self.token_prefix,  # [num_classes, 1, 768]
            self.ctx,           # [16, 768] -> [num_classes, 16, 768] (广播)
            self.token_suffix   # [num_classes, suffix_len, 768]
        ], dim=1)
        return prompts  # [num_classes, total_seq_len, 768]
```

#### B. 动态提示检索 (DynamicPromptRetriever)

```python
class DynamicPromptRetriever(nn.Module):
    def _retrieve_one_scale(self, query_feat, pool_feats_by_class, static_feats):
        """
        参数:
            query_feat: [1, 512] - 聚合后的图像特征
            pool_feats_by_class: list of [T_i, 512] - 每个类的句子特征池
            static_feats: [num_classes, 512] - 静态提示词特征
        
        返回:
            outputs: [num_classes, 512] - 检索融合后的特征
        """
        q = F.normalize(query_feat, dim=-1)  # [1, 512]
        outputs = []
        
        for c in range(self.num_classes):
            pool = pool_feats_by_class[c]  # [T_i, 512]
            
            # 计算相似度
            pool_n = F.normalize(pool, dim=-1)
            sim = torch.matmul(q, pool_n.t()).squeeze(0)  # [T_i]
            
            # 检索Top-K最相似的句子
            k = min(self.topk, sim.shape[0])
            top_vals, top_idx = torch.topk(sim, k=k, dim=0)  # 值和索引
            
            # 加权融合
            weights = F.softmax(top_vals / self.temperature, dim=0)  # [k]
            top_pool = pool[top_idx]  # [k, 512]
            retrieved = torch.sum(weights.unsqueeze(1) * top_pool, dim=0)  # [512]
            
            outputs.append(retrieved)
        
        return torch.stack(outputs, dim=0)  # [num_classes, 512]
```

## 训练循环

### 文件：`utils/core_utils.py:train_loop()`

```python
def train_loop(args, epoch, model, loader, optimizer, n_classes, writer, loss_fn, fold):
    
    model.train()
    acc_logger = Accuracy_Logger(n_classes=n_classes)
    train_loss = 0.0
    train_error = 0.0
    
    all_prob = []
    all_label = []
    all_pred = []
    
    for batch_idx, (data_s, coord_s, data_l, coords_l, label, slide_ids) in enumerate(loader):
        # 移至GPU
        data_s = data_s.to(device)      # [1, N_s, 512]
        coord_s = coord_s.to(device)
        data_l = data_l.to(device)      # [1, N_l, 512]
        coords_l = coords_l.to(device)
        label = label.to(device)        # [1]
        
        slide_id = slide_ids[0] if isinstance(slide_ids, (list, tuple)) else None
        
        # 前向传播
        Y_prob, Y_hat, loss = model(
            data_s, coord_s, data_l, coords_l, label, slide_id=slide_id
        )
        # Y_prob: [1, num_classes]
        # Y_hat: [1, 1]
        # loss: scalar
        
        # 反向传播和优化
        loss.backward()
        optimizer.step()
        optimizer.zero_grad()
        
        # 记录指标
        acc_logger.log(Y_hat, label)
        train_loss += loss.item()
        
        batch_error = calculate_error(Y_hat, label)
        train_error += batch_error
        
        all_prob.append(Y_prob.detach().cpu().numpy())
        all_label.append(label.cpu().numpy())
        all_pred.append(Y_hat.cpu().numpy())
    
    # 计算平均值
    train_loss /= len(loader)
    train_error /= len(loader)
    
    # 计算AUC和F1
    all_prob = np.concatenate(all_prob, axis=0)      # [n_samples, num_classes]
    all_label = np.concatenate(all_label, axis=0)    # [n_samples]
    
    if n_classes == 2:
        train_auc = roc_auc_score(all_label, all_prob[:, 1])
    else:
        train_auc = roc_auc_score(all_label, all_prob, multi_class='ovr')
    
    train_f1 = f1_score(all_label, all_pred.flatten(), average='weighted')
    
    return train_loss, train_error, train_auc, train_f1
```
